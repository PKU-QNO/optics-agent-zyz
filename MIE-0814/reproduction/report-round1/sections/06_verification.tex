\subsection{Eq.1 解析常数推导（第一性原理）}

将 $J=-i\omega\varepsilon_0\widetilde{J}$ 代回 Eq.~\eqref{eq:eq1}，再除以
$\lambda^2/(2\pi)$，逐项推导：

\textbf{矩的 $\varepsilon_0/c$ 尺度}（表2 定义带 $-1/i\omega$ 因子）：
\begin{align}
  p_{\mathrm{phys}} &= \varepsilon_0\,p_{\mathrm{tilde}}, &
  m_{\mathrm{phys}} &= -i\omega\varepsilon_0\,m_{\mathrm{tilde}}, \notag\\
  Q^e_{\mathrm{phys}} &= \varepsilon_0\,Q^e_{\mathrm{tilde}}, &
  Q^m_{\mathrm{phys}} &= -i\omega\varepsilon_0\,Q^m_{\mathrm{tilde}}.
  \label{eq:mom_scale}
\end{align}

逐项代入 + 归一化（$\lambda^2/(2\pi)=2\pi/k^2$）：

\begin{align}
  C_{\mathrm{ED}} &= \frac{k^6}{12\pi^2}|p_{\mathrm{tilde}}|^2
     = \frac{x^6}{12\pi^2}|p|^2, \label{eq:der_ed}\\
  C_{\mathrm{MD}} &= \frac{k^8}{12\pi^2}|m_{\mathrm{tilde}}|^2
     = \frac{x^8}{12\pi^2}|m|^2, \label{eq:der_md}\\
  C_{\mathrm{EQ}} &= \frac{k^8}{1440\pi^2}|Q^e_{\mathrm{tilde}}|^2
     = \frac{x^8}{1440\pi^2}|Q^e|^2, \label{eq:der_eq}\\
  C_{\mathrm{MQ}} &= \frac{k^{10}}{1440\pi^2}|Q^m_{\mathrm{tilde}}|^2
     = \frac{x^{10}}{1440\pi^2}|Q^m|^2. \label{eq:der_mq}
\end{align}

\textbf{关键点}：

\begin{enumerate}
  \item \textbf{$1/1440 = 1/(120\times 12)$}：其中 120 来自 Eq.~\eqref{eq:eq1}
        的四极系数 $1/120$；12 来自前因子 $k^4/(6\pi\varepsilon_0^2)$ 的 $6$
        与 $\lambda^2/(2\pi)=2\pi/k^2$ 归一化的 $2$ 的组合（$6\times2=12$）。
  \item \textbf{MQ 是 $x^{10}$ 而非 $x^8$}：磁四极在 Eq.~\eqref{eq:eq1} 带 $1/c^2$，
        该 $c^2$ 引入额外的 $k^2=x^2$ 因子（因 $m_{\mathrm{phys}},Q^m_{\mathrm{phys}}$
        含 $i\omega\propto k$，而 $p,Q^e$ 的 $-1/i\omega$ 与 $-i\omega\varepsilon_0$ 抵消）。
        这是最容易出错的地方。
  \item \textbf{MD 是 $x^8$ 而非 $x^6$}：同理，$|m|^2/c^2$ 的 $1/c^2$ 带来 $k^2$。
  \item ED 与 MD 偶极常数同为 $1/12\pi^2$；EQ 与 MQ 四极常数同为 $1/1440\pi^2$。
\end{enumerate}

\textbf{数值联合验证}：准静态点 $s=0.2,0.3$ 四通道表2/Mie 全部 $\approx1.0000$
（ED 1.0000、MD 1.0000、EQ 1.00003、MQ 0.99997）。EQ/MQ 在小 $x$ 是纯四极项，
直接锁定 $1/1440$ 常数正确。

\paragraph{关键说明（非程序错误）}
\textbf{表1 在大尺寸/共振区发散是论文预期的物理行为，不是程序错误。}
表1 是长波长近似（$kr\to0$ 极限），在大尺寸/共振区粒子内部相位不再一致，
近似必然失效。相反，表2 精确公式与 Mie 在应用近零掩码后 200 点最大误差均低于 $1\%$
（ED 0.1404\%、MD 0.0383\%、EQ 0.2020\%、MQ 0.1252\%）——图~\ref{fig:fig1a}
中表1 点线在大尺寸/共振区的发散正是论文 Fig.1(c) 想展示的近似失效边界。
